
                                                 For Google Chrome

set -eu

echo "==> Searching for Chrome"
for f in \
  /usr/bin/google-chrome \
  /usr/bin/google-chrome-stable \
  /opt/google/chrome/google-chrome \
  /opt/google/chrome/chrome
do
  if [ -x "$f" ]; then
    echo "Found: $f"
    "$f" --version
    break
  fi
done

if [ ! -x /opt/google/chrome/google-chrome ] &&
   [ ! -x /usr/bin/google-chrome-stable ] &&
   [ ! -x /usr/bin/google-chrome ]; then

  echo "==> Checking the downloaded package"
  if [ ! -s /tmp/google-chrome-stable.deb ]; then
    wget -O /tmp/google-chrome-stable.deb \
      https://dl.google.com/linux/direct/google-chrome-stable_current_amd64.deb
  fi

  echo "==> Installing Chrome"
  apt-get update
  apt-get install -y /tmp/google-chrome-stable.deb
fi

echo "==> Refreshing command lookup"
hash -r

CHROME=""
for f in /usr/bin/google-chrome /usr/bin/google-chrome-stable \
         /opt/google/chrome/google-chrome; do
  if [ -x "$f" ]; then
    CHROME="$f"
    break
  fi
done

if [ -z "$CHROME" ]; then
  echo "ERROR: Chrome executable is still missing."
  echo "Package status:"
  dpkg -l google-chrome-stable || true
  echo "Package file list:"
  dpkg -L google-chrome-stable 2>/dev/null | head -n 30 || true
  exit 1
fi

echo "==> Chrome installed at: $CHROME"
"$CHROME" --version

ln -sf "$CHROME" /usr/local/bin/google-chrome
echo "==> Test complete. Run Chrome with:"
echo "google-chrome --no-sandbox --disable-dev-shm-usage"


google-chrome --no-sandbox --disable-dev-shm-usage
